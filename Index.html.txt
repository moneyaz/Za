        function showToast(message, isError = false) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-msg');
            const toastIcon = document.getElementById('toast-icon');
            
            toastMsg.innerText = message;
            toastIcon.className = isError ? "fas fa-circle-exclamation text-xl text-red-400" : "fas fa-check-circle text-xl text-emerald-400";

            toast.classList.remove('translate-y-[-150%]', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-[-150%]', 'opacity-0');
            }, 3500);
        }

        function checkUrlInviteCode() {
            const urlParams = new URLSearchParams(window.location.search);
            const refParam = urlParams.get('ref');

            if (refParam) {
                document.getElementById('lock-screen').style.display = 'none';
                showToast("Bem-vindo ao MONEY AZ!");
            }
        }

        function unlockWithCode() {
            const input = document.getElementById('invite-code-input').value.trim().toUpperCase();
            if (VALID_INVITES.includes(input) || input.startsWith('VIP')) {
                document.getElementById('lock-screen').style.display = 'none';
                showToast("Acesso MONEY AZ liberado com sucesso!");
            } else {
                showToast("Código de convite inválido.", true);
            }
        }

        function formatKz(amount) {
            return new Intl.NumberFormat('pt-AO').format(amount) + ' Kz';
        }

        function renderDashboard() {
            document.getElementById('user-balance').innerText = formatKz(state.user.balance);
            document.getElementById('user-pending-balance').innerText = formatKz(state.user.pendingBalance);
            document.getElementById('user-referral-earnings').innerText = formatKz(state.user.referralEarnings);
            document.getElementById('user-referral-count').innerText = state.user.referralCount;

            const baseUrl = window.location.href.split('?')[0];
            document.getElementById('referral-link-input').value = `${baseUrl}?ref=VIP123`;

            const table = document.getElementById('user-transactions-table');
            table.innerHTML = '';

            const allActivities = [
                ...state.deposits.map(d => ({ date: d.date, name: 'Recarga de Saldo', amount: d.amount, status: d.status })),
                ...state.purchases.map(p => ({ date: p.date, name: 'Compra: ' + p.packageName, amount: -p.price, status: 'APROVADO' }))
            ].sort((a,b) => new Date(b.date) - new Date(a.date));

            if (allActivities.length === 0) {
                table.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-slate-500">Nenhuma atividade registada.</td></tr>`;
                return;
            }

            allActivities.forEach(act => {
                const tr = document.createElement('tr');
                const isPending = act.status === 'PENDENTE';
                const isNegative = act.amount < 0;

                tr.innerHTML = `
                    <td class="p-3 text-slate-400">${act.date}</td>
                    <td class="p-3 font-semibold text-white">${act.name}</td>
                    <td class="p-3 font-mono font-bold ${isNegative ? 'text-amber-400' : 'text-emerald-400'}">${isNegative ? '' : '+'}${formatKz(act.amount)}</td>
                    <td class="p-3">
                        <span class="px-2 py-0.5 rounded-md text-[10px] font-bold ${isPending ? 'bg-amber-500/10 text-amber-400 border border-amber-500/20' : 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20'}">
                            ${act.status}
                        </span>
                    </td>
                `;
                table.appendChild(tr);
            });
        }

        function renderPackages() {
            const grid = document.getElementById('packages-grid');
            grid.innerHTML = '';

            state.packages.forEach(pkg => {
                const card = document.createElement('div');
                card.className = "glass-card rounded-2xl p-5 border border-slate-800 flex flex-col justify-between relative hover:border-amber-500/50 transition-all duration-300";

                const isOutOfStock = pkg.stock <= 0;

                            ${pkg.badge ? `<span class="bg-amber-500/10 text-amber-400 border border-amber-500/30 text-[10px] font-bold px-2 py-0.5 rounded-full">${pkg.badge}</span>` : ''}
                        </div>
                        <div class="inline-block px-2.5 py-1 rounded-lg bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 text-xs font-bold mb-3">
                            Rendimento: 7,5% ao dia (${formatKz(pkg.dailyYield)}/dia)
                        </div>
                        <div class="text-2xl font-black text-amber-400 mb-2">${formatKz(pkg.price)}</div>
                        <div class="text-[11px] text-slate-400 mb-5 flex items-center justify-between">
                            <span>Unidades restantes:</span>
                            <span class="font-bold ${isOutOfStock ? 'text-red-400' : 'text-emerald-400'}">${pkg.stock} / 100</span>
                        </div>
                    </div>

                    <button onclick="buyPackage('${pkg.id}')" ${isOutOfStock ? 'disabled' : ''} class="w-full py-3 rounded-xl font-bold text-xs transition flex items-center justify-center gap-2 ${isOutOfStock ? 'bg-slate-800 text-slate-500 cursor-not-allowed' : 'bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-slate-950 shadow-md shadow-amber-500/10 active:scale-95'}">
                        <i class="fas fa-shopping-cart"></i> ${isOutOfStock ? 'Esgotado' : 'Comprar com Saldo'}
                    </button>
                `;
                grid.appendChild(card);
            });
        }

        function buyPackage(pkgId) {
            const pkg = state.packages.find(p => p.id === pkgId);
            if (!pkg) return;

            if (pkg.stock <= 0) {
                showToast("Este pacote está esgotado!", true);
                return;
            }

            if (state.user.balance < pkg.price) {
                showToast(`Saldo insuficiente. Recarregue ${formatKz(pkg.price - state.user.balance)}.`, true);
                openDepositModal();
                return;
            }

            state.user.balance -= pkg.price;
            pkg.stock -= 1;

            state.purchases.push({
                id: 'PUR-' + Date.now(),
                packageName: pkg.name,
                price: pkg.price,
                date: new Date().toLocaleDateString('pt-PT')
            });

            localStorage.setItem('moneyaz_packages', JSON.stringify(state.packages));
            localStorage.setItem('moneyaz_purchases', JSON.stringify(state.purchases));

            renderDashboard();
            renderPackages();
            showToast(`Adquiriu o ${pkg.name} com sucesso!`);
        }

        function submitDeposit(e) {
            e.preventDefault();
            const amount = parseFloat(document.getElementById('deposit-amount').value);
            const ref = document.getElementById('deposit-ref').value.trim();

            if (!amount || amount < 1000) {
                showToast("Valor mínimo para recarga: 1.000 Kz", true);
                return;
            }

            const newDeposit = {
                id: 'DEP-' + Date.now(),
                amount: amount,
                ref: ref,
                status: 'PENDENTE',
                date: new Date().toLocaleDateString('pt-PT')
            };

            state.deposits.push(newDeposit);
            state.user.pendingBalance += amount;

            localStorage.setItem('moneyaz_deposits', JSON.stringify(state.deposits));

            closeDepositModal();
            renderDashboard();
            showToast("Comprovativo submetido com sucesso! Aguarde aprovação.");
        }

        function renderAdminView() {
            const approvedDeposits = state.deposits.filter(d => d.status === 'APROVADO');
            const totalDeposited = approvedDeposits.reduce((acc, d) => acc + d.amount, 0);
            const pendingDeposits = state.deposits.filter(d => d.status === 'PENDENTE');

            document.getElementById('adm-stat-deposits').innerText = formatKz(totalDeposited);
            document.getElementById('adm-stat-sales').innerText = state.purchases.length;
            document.getElementById('adm-stat-pending').innerText = pendingDeposits.length;
            document.getElementById('pending-count-badge').innerText = pendingDeposits.length;

            const table = document.getElementById('admin-pending-table');
            table.innerHTML = '';

            if (pendingDeposits.length === 0) {
                table.innerHTML = `<tr><td colspan="4" class="p-3 text-center text-slate-500">Sem recargas pendentes.</td></tr>`;
            } else {
                pendingDeposits.forEach(dep => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td class="p-2.5 text-slate-400">${dep.date}</td>
                        <td class="p-2.5 font-mono text-white">${dep.ref}</td>
                        <td class="p-2.5 font-bold text-emerald-400">${formatKz(dep.amount)}</td>
                        <td class="p-2.5 flex gap-2">
                            <button onclick="approveDeposit('${dep.id}')" class="px-2.5 py-1 bg-emerald-600 text-white rounded font-bold hover:bg-emerald-500">Aprovar</button>
                            <button onclick="rejectDeposit('${dep.id}')" class="px-2.5 py-1 bg-red-600 text-white rounded font-bold hover:bg-red-500">Rejeitar</button>
                        </td>
                    `;
                    table.appendChild(tr);
                });
            }

            const stockList = document.getElementById('admin-stock-list');
            stockList.innerHTML = '';
            state.packages.forEach(p => {
                const item = document.createElement('div');
                item.className = "flex justify-between items-center bg-slate-900 p-2.5 rounded-lg border border-slate-800 text-xs";
                item.innerHTML = `
                    <div>
                        <span class="font-bold text-white">${p.name}</span>
                        <span class="text-slate-400 block">${formatKz(p.price)} (Rendimento: ${formatKz(p.dailyYield)}/dia)</span>
                    </div>
                    <div class="flex items-center gap-2">
                        <span class="text-slate-400">Stock:</span>
                        <input type="number" value="${p.stock}" min="0" onchange="updatePackageStock('${p.id}', this.value)" class="w-16 px-2 py-1 bg-slate-800 border border-slate-700 rounded text-center text-amber-400 font-bold">
                    </div>
                `;
                stockList.appendChild(item);
            });
        }

        function approveDeposit(depId) {
            const dep = state.deposits.find(d => d.id === depId);
            if (!dep) return;

            dep.status = 'APROVADO';
            state.user.pendingBalance -= dep.amount;
            state.user.balance += dep.amount;

            state.user.referralEarnings += dep.amount * 0.10;
            state.user.referralCount += 1;

            localStorage.setItem('moneyaz_deposits', JSON.stringify(state.deposits));

            renderAdminView();
            renderDashboard();
            showToast(`Recarga de ${formatKz(dep.amount)} APROVADA!`);
        }

        function rejectDeposit(depId) {
            const dep = state.deposits.find(d => d.id === depId);
            if (!dep) return;

            dep.status = 'REJEITADO';
            state.user.pendingBalance -= dep.amount;

            localStorage.setItem('moneyaz_deposits', JSON.stringify(state.deposits));

            renderAdminView();
            renderDashboard();
            showToast("Recarga rejeitada.", true);
        }

        function updatePackageStock(pkgId, newStock) {
            const pkg = state.packages.find(p => p.id === pkgId);
            if (pkg) {
                pkg.stock = parseInt(newStock) || 0;
                localStorage.setItem('moneyaz_packages', JSON.stringify(state.packages));
                renderPackages();
                showToast(`Stock do ${pkg.name} atualizado.`);
            }
        }
